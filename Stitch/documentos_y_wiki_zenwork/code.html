// Inicialización del stream reactivo ZenWork
export async function initSyncWorker(projectId: string): Promise<WebSocketStream> {
  const endpoint = `wss://api.zenwork.internal/v1/projects/${projectId}/events`;
  const connection = new WebSocket(endpoint);

  connection.onopen = () => {
    connection.send(JSON.stringify({
      event: 'SUB_KANBAN_DELTA',
      workspace: 'ZEN-PROD',
      heartbeatIntervalMs: 15000
    }));
  };

  return connection;
}
